  </form>
<script>
let session;
const parentOrigin = new URLSearchParams(location.search).get('parent_origin');
if (!parentOrigin) throw new Error('parent_origin_required');
window.parent.postMessage({ type: 'pci-frame-ready' }, parentOrigin);
window.addEventListener('message', (evt) => {
  if (evt.origin !== parentOrigin) return;
  if (evt.data?.type === 'pci-session-init') session = evt.data.payload;
});
document.getElementById('pay').addEventListener('submit', async (e) => {
  e.preventDefault();
  if (!session) return;
  try {
    const res = await fetch(`${session.apiBaseUrl}/checkout/confirm`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${session.sessionToken}`,
        'X-Transaction-Id': session.transactionId,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ payment_method_token: 'pm_test_4242' }),
    });
    const data = await res.json();
    window.parent.postMessage({
      type: res.ok ? 'pci-checkout-success' : 'pci-checkout-error',
      payload: data,
    }, parentOrigin);
  } catch (err) {
    window.parent.postMessage({ type: 'pci-checkout-error', payload: { error: String(err) } }, parentOrigin);
  }
});
</script>
