<script type="module">
import {PciCheckout} from '/sdk/pci-checkout.js';
const BACKEND='http://localhost:3002';
const status=document.querySelector('#status');

function safeResult(data){
  if(!data || typeof data!=='object') return data;
  const copy={...data};
  delete copy.ephemeral_token;
  delete copy.client_secret;
  return copy;
}

document.querySelector('#start').onclick=async()=>{
  status.textContent='Criando checkout session via Merchant Backend...';
  const demo=await fetch(`${BACKEND}/api/demo/token`).then(r=>r.json());
  const r=await fetch(`${BACKEND}/api/checkout/session`,{
    method:'POST',
    headers:{Authorization:`Bearer ${demo.token}`,'Content-Type':'application/json'},
    body:JSON.stringify({amount:1099,currency:'brl',merchant_reference:`order_${Date.now()}`,return_url:location.origin+'/success'})
  });
  const session=await r.json();
  if(!r.ok){status.textContent=JSON.stringify(session,null,2);return;}
  status.textContent=`Session ${session.id} criada. O pagamento será enviado diretamente pelo iframe PCI ao API Gateway.`;

  const checkout=new PciCheckout({iframeUrl:session.iframe_url,target:document.querySelector('#secure')}).mount();
  checkout.configure({
    sessionId:session.id,
    transactionId:session.transaction_id,
    ephemeralToken:session.ephemeral_token || session.client_secret,
    paymentUrl:session.payment_api_url
  }).onResult(event=>{
    if(event.data?.type!=='pci:payment-result') return;
    status.textContent=JSON.stringify({http_status:event.data.httpStatus,...safeResult(event.data.result)},null,2);
  });
};
</script></body></html>
