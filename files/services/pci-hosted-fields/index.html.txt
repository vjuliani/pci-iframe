<script>
let cfg=null;
const parentOrigin='http://localhost:8880';
const msg=document.querySelector('#msg');

window.addEventListener('message',e=>{
  if(e.origin!==parentOrigin || e.data?.type!=='pci:init') return;
  const paymentUrl=String(e.data.paymentUrl||'');
  if(!paymentUrl.startsWith('http://') && !paymentUrl.startsWith('https://')) return;
  cfg={
    sessionId:String(e.data.sessionId||''),
    transactionId:String(e.data.transactionId||''),
    ephemeralToken:String(e.data.ephemeralToken||''),
    paymentUrl
  };
  document.querySelector('#pay').disabled=!(cfg.transactionId && cfg.ephemeralToken);
});

document.querySelector('#pay').onclick=async()=>{
  if(!cfg)return;
  const button=document.querySelector('#pay');
  button.disabled=true;
  msg.textContent='Tokenizando e enviando pagamento pela camada de autorização...';
  const pan=document.querySelector('#pan').value.replace(/\s/g,'');

  // POC only: simula tokenização dentro do domínio PCI. PAN/CVC nunca são
  // enviados ao merchant nem ao API Gateway; somente o payment_method_token.
  const paymentMethodToken=`pm_mock_${pan.slice(-4)}_${crypto.randomUUID().slice(0,8)}`;

  let httpStatus=0, result={};
  try {
    const response=await fetch(cfg.paymentUrl,{
      method:'POST',
      headers:{
        'Authorization':`Bearer ${cfg.ephemeralToken}`,
        'X-Transaction-Id':cfg.transactionId,
        'Content-Type':'application/json'
      },
      body:JSON.stringify({payment_method_token:paymentMethodToken})
    });
    httpStatus=response.status;
    try { result=await response.json(); } catch { result={error:'invalid_gateway_response'}; }
  } catch (error) {
    result={error:'payment_gateway_unreachable',reason:String(error?.message||error)};
  }

  if(httpStatus===200){
    msg.textContent='Pagamento processado.';
  }else if(httpStatus===403 && result?.error==='authorization_context_missing'){
    msg.textContent='Pagamento não processado: o ambiente local não executou a camada de autorização.';
  }else{
    msg.textContent=`Pagamento retornou HTTP ${httpStatus||'network-error'}.`;
  }
  window.parent.postMessage({type:'pci:payment-result',httpStatus,result},parentOrigin);
  button.disabled=false;
};
</script></body></html>
